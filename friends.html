/* ========================= LOAD FRIENDS ========================= */ async function loadFriends(){ const box = document.getElementById("friends"); if(!box){ console.error("Friends box not found"); return; } /* ========================= CHECK LOGIN ========================= */ if(!currentUser){ box.innerHTML = `
لطفاً ابتدا وارد حساب شوید.
`; return; } /* ========================= LOAD FRIENDSHIPS ========================= */ const { data: friendships, error: friendshipError } = await supabase .from("friendships") .select("user_id, friend_id") .or( `user_id.eq.${currentUser.id},friend_id.eq.${currentUser.id}` ); if(friendshipError){ console.error( "Friendship error:", friendshipError ); box.innerHTML = `
خطا در دریافت دوستان.
`; return; } /* ========================= NO FRIENDS ========================= */ if( !friendships || friendships.length === 0 ){ box.innerHTML = `
هنوز دوستی نداری.
`; return; } /* ========================= FIND FRIEND IDS ========================= */ const friendIds = []; friendships.forEach(friendship => { let friendId; if( friendship.user_id === currentUser.id ){ friendId = friendship.friend_id; }else{ friendId = friendship.user_id; } if( friendId && friendId !== currentUser.id ){ friendIds.push(friendId); } }); /* حذف ID های تکراری */ const uniqueFriendIds = [...new Set(friendIds)]; if(uniqueFriendIds.length === 0){ box.innerHTML = `
هنوز دوستی نداری.
`; return; } /* ========================= LOAD PROFILES ========================= */ const { data: profiles, error: profileError } = await supabase .from("profiles") .select("id, username") .in( "id", uniqueFriendIds ); if(profileError){ console.error( "Profile error:", profileError ); box.innerHTML = `
خطا در دریافت نام دوستان.
`; return; } /* ========================= CREATE PROFILE MAP ========================= */ const profileMap = new Map( (profiles || []).map(profile => [ profile.id, profile.username ]) ); /* ========================= CLEAR FRIENDS ========================= */ box.innerHTML = ""; /* ========================= DISPLAY FRIENDS ========================= */ uniqueFriendIds.forEach( friendId => { const username = profileMap.get(friendId); const div = document.createElement("div"); div.className = "friend"; div.innerHTML = `
👤 ${ escapeHTML( username || "نام کاربر پیدا نشد" ) }
FRIEND
`; box.appendChild(div); } ); }
