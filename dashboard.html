saveBtn.addEventListener("click", async function () { if (!currentUser) { alert("ابتدا وارد حساب خود شوید."); return; } const newUsername = usernameInput.value.trim(); if (!newUsername) { alert("لطفاً اسم گیم خود را وارد کنید."); return; } if (newUsername.length < 2) { alert("اسم گیم باید حداقل ۲ کاراکتر باشد."); return; } saveBtn.disabled = true; saveBtn.textContent = "در حال ذخیره..."; try { const { data, error } = await supabase .from("profiles") .update({ username: newUsername }) .eq("id", currentUser.id) .select("id, username"); console.log("USER ID:", currentUser.id); console.log("DATA:", data); console.log("ERROR:", error); if (error) { throw error; } if (!data || data.length === 0) { throw new Error( "هیچ پروفایلی برای این حساب پیدا نشد." ); } usernameText.textContent = data[0].username; profileUsername.textContent = data[0].username; usernameInput.value = data[0].username; showStatus( "اسم گیم با موفقیت ذخیره شد ✅", "success" ); } catch (error) { console.error( "SAVE ERROR:", error ); alert( "❌ ذخیره نشد\n\n" + "خطا:\n" + error.message ); showStatus( "ذخیره انجام نشد", "error" ); } saveBtn.disabled = false; saveBtn.textContent = "ثبت تغییر"; });
